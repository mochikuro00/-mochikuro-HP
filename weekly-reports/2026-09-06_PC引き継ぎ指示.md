### 3-1. アポ管理へ追記（GAS 推奨・重複防止つき）
営業管理シートの Apps Script に下記を貼って `appendApoFromStaging_20260906()` を1回実行。既存行と「アポ日＋相手名」が同じものは飛ばす。

```javascript
function appendApoFromStaging_20260906() {
  var STAGING_ID = '1Je5LCjHW8dcf26cBrwJCveZ153xPN-A3lZqK9M4eoNY';
  var src = SpreadsheetApp.openById(STAGING_ID).getSheets()[0];
  var dst = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('アポ管理');
  if (!dst) throw new Error('アポ管理 タブが見つかりません');

  var srcVals = src.getDataRange().getValues();           // 1行目ヘッダー
  var last = dst.getLastRow();
  var existing = last >= 3 ? dst.getRange(3, 1, last - 2, 4).getValues() : [];
  var seen = {};
  existing.forEach(function (r) {
    var d = r[0] instanceof Date ? Utilities.formatDate(r[0], 'JST', 'yyyy/MM/dd') : String(r[0]);
    seen[d + '|' + String(r[3]).trim()] = true;
  });

  var out = [];
  for (var i = 1; i < srcVals.length; i++) {
    var r = srcVals[i];
    if (!r[0] || !r[3]) continue;
    var dt = r[0] instanceof Date ? r[0] : new Date(String(r[0]).replace(/-/g, '/'));
    var key = Utilities.formatDate(dt, 'JST', 'yyyy/MM/dd') + '|' + String(r[3]).trim();
    if (seen[key]) continue;
    seen[key] = true;
    var row = r.slice(0, 11);
    while (row.length < 11) row.push('');
    row[0] = new Date(dt.getFullYear(), dt.getMonth(), dt.getDate());   // Date型で書く
    if (row[7] && !(row[7] instanceof Date)) row[7] = String(row[7]);
    out.push(row);
  }
  if (out.length) {
    dst.getRange(dst.getLastRow() + 1, 1, out.length, 11).setValues(out);
    dst.getRange(dst.getLastRow() - out.length + 1, 1, out.length, 1).setNumberFormat('yyyy/mm/dd');
  }
  Logger.log('追記 ' + out.length + ' 行（重複スキップ ' + (srcVals.length - 1 - out.length) + ' 行）');
  if (typeof markDirty_ === 'function') markDirty_();
}
```
